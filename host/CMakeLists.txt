cmake_minimum_required(VERSION 3.16)
project(dualmesh_host CXX)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -O2)
include_directories(../lib/dualmesh/src sim)
add_library(scheduler ../lib/dualmesh/src/radio_scheduler.cpp)
add_library(presets ../lib/dualmesh/src/presets_meshtastic.cpp
                    ../lib/dualmesh/src/presets_meshcore.cpp)
add_executable(sim sim/sim_main.cpp)
target_link_libraries(sim scheduler presets)
add_executable(test_phy tests/test_phy.cpp)
add_executable(test_presets tests/test_presets.cpp)
target_link_libraries(test_presets presets)
add_library(diag ../lib/dualmesh/src/diag_report.cpp)
add_executable(test_report tests/test_report.cpp)
target_link_libraries(test_report diag)
add_executable(test_i18n tests/test_i18n.cpp ../lib/dualmesh/src/i18n.cpp)
